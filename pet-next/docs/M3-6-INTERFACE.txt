新增 core/feedback/types.ts、reduce.ts（或 index.ts 汇总），只导出：

  export type FeedbackPhase = 'idle' | 'waiting' | 'hint';
  export interface FeedbackState {
    epoch: number;            // reset 递增；旧 epoch 事件一律无效
    token: number;            // 内部递增，只在 reduce 内产生；对外不暴露
    phase: FeedbackPhase;
    drag: boolean;            // 拖动优先
    /** 当前提示的隐藏期限（注入时钟）；无提示为 null。任意时刻至多一个定时器。 */
    deadlineAtMs: number | null;
    /** 该成功是否已消耗过回应（去重：每 token 至多一次 react）。 */
    reacted: boolean;
    /** dispose 之后任何事件零效果（负责人补充字段，B 已实现）。 */
    disposed: boolean;
  }
  export type FeedbackEvent =
    | { type: 'submit' }                                  // 有效提交开始（新 token）
    | { type: 'ok'; token: number }
    | { type: 'error'; token: number }
    | { type: 'cancel'; token: number }
    | { type: 'reset' }                                   // 新对话：清展示+旧 token 全体失效
    | { type: 'dispose' }
    | { type: 'drag-start' } | { type: 'drag-end' }
    | { type: 'expire'; token: number };                   // 调用方定时器到期回灌
  export type FeedbackEffect =
    | { type: 'hint-show'; kind: 'waiting' | 'success' | 'error' | 'cancelled' }
    | { type: 'hint-hide'; reason: 'expire' | 'reset' | 'dispose' }
    | { type: 'react'; token: number };                    // 请求一次回应（走 behavior）
  export function initialFeedbackState(): FeedbackState;
  export function reduceFeedback(
    state: FeedbackState,
    event: FeedbackEvent,
    options: { nowMs: number; hideAfterMs: number },
  ): { state: FeedbackState; effects: FeedbackEffect[] };

语义硬约束（逐条有单测）：
  - submit → 新 token、waiting 提示、deadline=now+hideAfterMs
  - ok(当前 token)：hint-show 'success'；≤1 次 react；reacted=true。非当前 token（迟到/被 reset/被 cancel）→ 零 effects
  - error(当前 token)：hint-show 'error'（不 react、不假成功）；非当前 token → 零 effects
  - cancel(当前 token)：hint-show 'cancelled'；非当前 → 零 effects
  - 重复终态（同 token 二次 ok/error/cancel）→ 零 effects
  - reset：epoch++、token 失效、phase→idle、deadline→null；effects=[hint-hide(reset)]
  - dispose：同上 + hint-hide(dispose)；之后任何事件零 effects
  - drag-start：drag=true；如当前有 waiting/hint → 保留 waiting 但挂起 react 资格（不排长队）
  - drag 中 ok：hint-show 'success'，不 react（丢弃动作，保留结果提示）
  - drag-end：drag=false；不补发被丢弃的 react
  - expire(当前 token 且 nowMs>=deadline)：hint-hide(expire)；否则零 effects
  - 连续 submit：旧 token 立即失效（其终态零 effects）
